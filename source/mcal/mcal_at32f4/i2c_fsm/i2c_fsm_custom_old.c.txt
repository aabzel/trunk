static bool i2c_fsm_all_flag_clear(I2cHandle_t* const Node) {
	bool res = true;
    // I2C_BUSERR_FLAG: bus error
	i2c_flag_clear(Node->I2Cx, I2C_BUSERR_FLAG);
    // I2C_ARLOST_FLAG: arbitration lost
	i2c_flag_clear(Node->I2Cx, I2C_ARLOST_FLAG);
    // I2C_ACKFAIL_FLAG: acknowledge failure
	i2c_flag_clear(Node->I2Cx, I2C_ACKFAIL_FLAG);
    // I2C_OUF_FLAG: overflow or underflow
	i2c_flag_clear(Node->I2Cx, I2C_OUF_FLAG);
    // I2C_PECERR_FLAG: pec receive error
	i2c_flag_clear(Node->I2Cx, I2C_PECERR_FLAG);
    // I2C_TMOUT_FLAG: smbus timeout
	i2c_flag_clear(Node->I2Cx, I2C_TMOUT_FLAG);
    // I2C_ALERTF_FLAG: smbus alert
	i2c_flag_clear(Node->I2Cx, I2C_ALERTF_FLAG);
    // I2C_STOPF_FLAG: stop condition generation complete
	i2c_flag_clear(Node->I2Cx, I2C_STOPF_FLAG);
    // I2C_ADDR7F_FLAG: i2c 0~7 bit address match
	i2c_flag_clear(Node->I2Cx, I2C_ADDR7F_FLAG);
	return res;
}

static bool i2c_fsm_inpit_smbus_get(I2cHandle_t* const Node) {
    bool res = false;
    //  I2C_TMOUT_FLAG: smbus timeout
    if(false==Node->fsm_input_exist) {

        ret = i2c_flag_get(Node->I2Cx, I2C_TMOUT_FLAG);
        if(SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_SMBUS_TIMEOUT;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist=true;
            res = true;
        }
    }

    if(false==Node->fsm_input_exist) {

        //  I2C_ALERTF_FLAG: smbus alert
        ret = i2c_flag_get(Node->I2Cx, I2C_ALERTF_FLAG);
        if(SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_SMBUS_ALERT;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist=true;
            res = true;
        }
    }

    if(false==Node->fsm_input_exist) {

        //  I2C_DEVADDRF_FLAG: smbus device address received
        ret = i2c_flag_get(Node->I2Cx, I2C_DEVADDRF_FLAG);
        if(SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_SMBUS_DEVICE_ADDRESS_RX;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist=true;
            res = true;
        }
    }

    if(false==Node->fsm_input_exist) {

        //  I2C_HOSTADDRF_FLAG: smbus host address received
        ret = i2c_flag_get(Node->I2Cx, I2C_HOSTADDRF_FLAG);
        if(SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_SMBUS_HOST_ADDRESS_RX;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist=true;
            res = true;
        }
    }
    return res;
}

#define I2C_FSM_GET_FLAG(FLAG, ENUM)                                                                                   \
    do {                                                                                                               \
        ret = i2c_flag_get(Node->I2Cx, FLAG);                                                                          \
        if(SET == ret) {                                                                                               \
            Node->fsm_input = ENUM;                                                                                    \
            Node->fsm_input_cnt++;                                                                                     \
            res = true;                                                                                                \
        }                                                                                                              \
    } while(0);

bool i2c_fsm_inpit_get(I2cHandle_t *const Node) {
    bool res = false;
    flag_status ret = 0;
    Node->fsm_input_cnt = 0;

    if (false == Node->fsm_input_exist) {
        //   I2C_STOPF_FLAG: stop condition generation complete
        ret = i2c_flag_get(Node->I2Cx, I2C_STOPF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_STOP_SIGNAL_DONE;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        // i2c 0~7 bit address match flag
        ret = i2c_flag_get(Node->I2Cx, I2C_ADDR7F_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_7BITADDR_ACK;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_ACKFAIL_FLAG: acknowledge failure
        ret = i2c_flag_get(Node->I2Cx, I2C_ACKFAIL_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_ACKNOWLEDGE_FAIL;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_STARTF_FLAG: start condition generation complete
        ret = i2c_flag_get(Node->I2Cx, I2C_STARTF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_START_DONE;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_TDBE_FLAG: transmit data buffer empty
        ret = i2c_flag_get(Node->I2Cx, I2C_RDBF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_TX_DATA_BUFF_EMPTY;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_ADDRHF_FLAG: master 9~8 bit address header match
        ret = i2c_flag_get(Node->I2Cx, I2C_ADDRHF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_ADDRH_9_8_ACK;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_RDBF_FLAG: receive data buffer full
        ret = i2c_flag_get(Node->I2Cx, I2C_RDBF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_RX_DATA_FULL;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_ARLOST_FLAG: arbitration lost
        ret = i2c_flag_get(Node->I2Cx, I2C_ARLOST_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_ARBITRATION_LOST;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_OUF_FLAG: overflow or underflow
        ret = i2c_flag_get(Node->I2Cx, I2C_OUF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_OVERFLOW_OR_UNDERFLOW;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_PECERR_FLAG: pec receive error
        ret = i2c_flag_get(Node->I2Cx, I2C_PECERR_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_RX_PEC;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_GCADDRF_FLAG: general call address received
        ret = i2c_flag_get(Node->I2Cx, I2C_GCADDRF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_GENERAL_CALL_ADDRESS_RECEIVED;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_ADDR2_FLAG: own address 2 received
        ret = i2c_flag_get(Node->I2Cx, I2C_ADDR2_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_OWN_ADDRESS_2_RX;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_DIRF_FLAG: transmission direction
        ret = i2c_flag_get(Node->I2Cx, I2C_DIRF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_TX_DIR;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_TDC_FLAG: transmit data complete
        ret = i2c_flag_get(Node->I2Cx, I2C_TDC_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_TX_DATA_DONE;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_BUSYF_FLAG: bus busy flag transmission mode.
        ret = i2c_flag_get(Node->I2Cx, I2C_BUSYF_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_BUS_BUSY;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_BUSERR_FLAG: bus error
        ret = i2c_flag_get(Node->I2Cx, I2C_BUSERR_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_BUS_ERROR;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

    if (false == Node->fsm_input_exist) {
        //  I2C_TRMODE_FLAG: transmission mode.
        ret = i2c_flag_get(Node->I2Cx, I2C_TRMODE_FLAG);
        if (SET == ret) {
            Node->fsm_input = I2C_FSM_INPUT_TX_MODE;
            Node->fsm_input_cnt++;
            Node->fsm_input_exist = true;
            res = true;
        }
    }

#ifdef HAS_SMBUS
    res = i2c_fsm_inpit_smbus_get(Node);
#endif

    if (1 < Node->fsm_input_cnt) {
        LOG_ERROR(I2C, "I2C%u,MultIn:%u", Node->num, Node->fsm_input_cnt);
    } else if (1 == Node->fsm_input_cnt) {
        LOG_NOTICE(I2C, "I2C%u,In:%s", Node->num, I2cFsmInputToStr(Node->fsm_input));
    }
    return res;
}
